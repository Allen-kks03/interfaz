section .data
    ; Arreglo con los números primos entre 1 y 20
    primes db 2, 3, 5, 7, 11, 13, 17, 19

    ; Representaciones en 7 segmentos de los números del 0 al 9
    ; Cada valor corresponde a la configuración de los 7 segmentos para formar un número.
    display_digits db 0x3F, 0x06, 0x5B, 0x4F, 0x66, 0x6D, 0x7D, 0x07, 0x7F, 0x6F
    ; 0x3F -> 0, 0x06 -> 1, 0x5B -> 2, 0x4F -> 3, etc.

section .text
    global _start

_start:
    ; Inicializamos el índice para recorrer los números primos y el contador de primos
    mov si, 0             ; Índice para los números primos (inicio en 0)
    mov cx, 8             ; Número de primos (hay 8 números primos entre 1 y 20)

print_primes:
    ; Cargar el número primo actual en AL (registro de 8 bits)
    mov al, [primes + si]

    ; Verificar si el número es mayor que 9 (números de dos dígitos como 11, 13, etc.)
    cmp al, 9             ; Si AL > 9, es un número de dos dígitos
    jg show_two_digits    ; Si es mayor que 9, se llama a show_two_digits

    ; Si el número es de un solo dígito, se envía directamente al display
    mov bl, al            ; Copiar el número en el registro BL
    call display_digit    ; Mostrar el número en el display de 7 segmentos
    inc si                ; Aumentar el índice para mostrar el siguiente número primo
    loop print_primes      ; Decrementar CX y repetir si es mayor que 0

    jmp end_program       ; Si se completaron los 8 primos, terminamos el programa

show_two_digits:
    ; Si el número tiene dos dígitos, dividirlo en decenas y unidades
    mov dl, al            ; Copiar el número a DL
    div byte 10           ; Dividir AX (AL es el número) entre 10 -> AH = decenas, AL = unidades
    mov dl, ah            ; Decenas están ahora en AH, las copiamos a DL
    call display_digit    ; Mostrar el dígito de las decenas

    ; Mostrar el dígito de las unidades
    mov dl, al            ; El residuo de la división está en AL (unidades)
    call display_digit    ; Mostrar el dígito de las unidades

    inc si                ; Aumentar el índice para el siguiente número primo
    loop print_primes      ; Decrementar CX y repetir si es mayor que 0

display_digit:
    ; Convertir el número (0-9) a su representación en 7 segmentos
    mov al, [display_digits + dl] ; Obtenemos el valor en 7 segmentos para el dígito en DL
    out 0x378, al         ; Enviar la representación en 7 segmentos al puerto 0x378
    ret                   ; Volver a la rutina que llamó a display_digit

end_program:
    ; Detener la ejecución del programa
    hlt                   ; Fin del programa

Explicacion de la lógica

La lógica del código se basa en mostrar los números primos entre 1 y 20 en un display de 7 segmentos. 

1. **Datos**:
   - Se define un arreglo con los números primos (`2, 3, 5, 7, 11, 13, 17, 19`).
   - Otro arreglo contiene las representaciones en 7 segmentos de los números del 0 al 9.

2. **Recorrido de los primos**:
   - El programa recorre los números primos. Si el número es de un solo dígito, se muestra directamente en el display.
   - Si el número es de dos dígitos (como 11, 13, etc.), se divide en decenas y unidades y ambos dígitos se muestran secuencialmente.

3. **Conversión a 7 segmentos**:
   - Cada número se convierte a su correspondiente patrón en 7 segmentos utilizando el arreglo `display_digits`.
   
4. **Salida**:
   - Los valores de 7 segmentos se envían al puerto `0x378`, que controla el display (simulado en este caso como un puerto paralelo).

El programa repite este proceso hasta haber mostrado todos los números primos entre 1 y 20.
